```typescript
@Component
struct SceneCard {
  @ObjectLink scene: SceneModel;
  @State isPressed: boolean = false;
  @State glowOpacity: number = 0;

  build() {
    Column() {
      Image(this.scene.icon)
        .width(48)
        .height(48)
      Text(this.scene.name)
        .fontSize(14)
        .fontColor(this.scene.active ? '#007DFF' : '#E6000000')
        .margin({ top: 8 })
    }
    .width(120)
    .height(120)
    .backgroundColor(this.scene.active ? 'rgba(0, 125, 255, 0.1)' : '#F2F5F7')
    .borderRadius(16)
    .border({
      width: this.scene.active ? 1 : 0,
      color: this.scene.active ? '#33007DFF' : 'transparent'
    })
    // 核心光感：外发光叠加层
    .shadow({
      radius: this.scene.active ? 15 : 0,
      color: this.scene.active ? '#33007DFF' : 'transparent',
      offsetX: 0,
      offsetY: 0
    })
    .scale({ x: this.isPressed ? 0.97 : 1, y: this.isPressed ? 0.97 : 1 })
    .animation({ duration: 150, curve: Curve.EaseOut })
    .onTouch((event: TouchEvent) => {
      if (event.type === TouchType.Down) {
        this.isPressed = true;
        if (!this.scene.active) {
          // Hover 光感爆发动画
          animateTo({ duration: 100 }, () => { this.glowOpacity = 1; });
        }
      } else if (event.type === TouchType.Up || event.type === TouchType.Cancel) {
        this.isPressed = false;
        animateTo({ duration: 200 }, () => { this.glowOpacity = 0; });
      }
    })
    .gesture(
      TapGesture()
        .onAction(() => {
          this.scene.activate();
        })
    )
  }
}
```

```typescript
@Component
struct HaloSlider {
  @Link value: number; // 0-100
  @State sliderX: number = 0;
  @State isDragging: boolean = false;
  @State thumbScale: number = 1;
  private min: number = 0;
  private max: number = 100;

  aboutToAppear() {
    this.sliderX = 300; // 默认轨道宽度
  }

  private get thumbOffset(): number {
    const percent = (this.value - this.min) / (this.max - this.min);
    return percent * this.sliderX;
  }

  build() {
    Stack() {
      // 底层滑轨背景
      Row() {
        Rect()
          .width('100%')
          .height(6)
          .fill('#E6E6E6')
          .radius(3)
      }

      // 已填充进度（光感渐变）
      Row() {
        Rect()
          .fill(
            new LinearGradient({
              direction: GradientDirection.Left,
              colors: [['#007DFF', 0], ['#00A5FF', 1]]
            })
          )
          .height(6)
          .radius(3)
          .width(this.thumbOffset)
      }
      .clip(true)

      // 滑块（光晕效果）
      Circle()
        .width(this.isDragging ? 28 : 24)
        .height(this.isDragging ? 28 : 24)
        .fill('#007DFF')
        .shadow({
          radius: this.isDragging ? 20 : 10,
          color: '#33007DFF',
          offsetX: 0,
          offsetY: 4
        })
        .translate({ x: this.thumbOffset - (this.isDragging ? 14 : 12) })
        .animation({ duration: 150, curve: Curve.EaseOut })
    }
    .width('100%')
    .height(40)
    .gesture(
      PanGesture()
        .onActionStart(() => {
          this.isDragging = true;
          this.thumbScale = 1.15;
        })
        .onActionUpdate((event: GestureEvent) => {
          const newOffset = Math.max(0, Math.min(this.sliderX, event.localX));
          this.value = Math.floor(
            this.min + (newOffset / this.sliderX) * (this.max - this.min)
          );
        })
        .onActionEnd(() => {
          this.isDragging = false;
          this.thumbScale = 1;
          this.syncToDevice();
        })
    )
  }

  private syncToDevice(): void {

  }
}
```
